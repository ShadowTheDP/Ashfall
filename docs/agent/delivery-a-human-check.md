python -c "import os,subprocess;from pathlib import Path;p=Path('output/manual-delivery-a');p.mkdir(parents=True,exist_ok=True);env=dict(os.environ,ASHFALL_PROFILE_PATH=str((p/'profile.json').resolve()));log=(p/'play.log').open('a',encoding='utf-8');print('Human play profile:',env['ASHFALL_PROFILE_PATH']);subprocess.run([str(Path('.tools/Godot_v4.7.2-stable_win64.exe').resolve()),'--path',str(Path.cwd())],env=env,stdout=log,stderr=subprocess.STDOUT)"
